03 — 사용 코드

<!-- 1. SDK 추가 (head 또는 body 끝) -->
<script src="https://unvul-pov.dev/sdk.js"
       api-key="upov_xxx"></script>

<!-- 2. 이미지 태그에 unvul:// URL 사용 -->
<img src="unvul://f/암호화된데이터" />

<!-- 3. 동영상도 동일 -->
<video src="unvul://f/암호화된데이터" controls />

// JS에서 수동으로 URL 생성
const url = await window.unvul.encode({
  url: 'https://drive.google.com/uc?id=xxx',
  name: 'photo.jpg',
  type: 'image/jpeg'
});

// JS에서 수동으로 복호화
const meta = await window.unvul.resolve('unvul://f/...');
console.log(meta.url); // 실제 Drive URL
